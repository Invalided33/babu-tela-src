// Connections over the internet without a rented server, using Tailscale as
// a library (tsnet), so nobody installs Tailscale itself:
//
//   - The host pastes a Tailscale auth key once. Their app joins their
//     tailnet and serves the chat server and the TURN relay there.
//   - "Gerar convite" packs that key and the host's tailnet address into a
//     code. A friend's app joins the same tailnet with it and opens local
//     ports (127.0.0.1:3211 for the server, :3479 for TURN) that forward
//     through the tunnel, so the page and the calls work unchanged.
//
// The invite carries the auth key: anyone holding it can join the tailnet.

const (
	invitePrefix   = "BT1-"
	guestHTTPAddr  = "127.0.0.1:3211"
	guestTURNAddr  = "127.0.0.1:3479"
	guestServerURL = "http://" + guestHTTPAddr
)

type invite struct {
	Key  string `json:"k"`
	Host string `json:"h"` // host's tailnet IPv4
}

func encodeInvite(inv invite) string {
	b, _ := json.Marshal(inv)
	return invitePrefix + base64.RawURLEncoding.EncodeToString(b)
}

func decodeInvite(code string) (invite, error) {
	var inv invite
	code = strings.Join(strings.Fields(code), "")
	if !strings.HasPrefix(code, invitePrefix) {
		return inv, errors.New("Esse não é um convite do Babu Tela.")
	}
	b, err := base64.RawURLEncoding.DecodeString(strings.TrimPrefix(code, invitePrefix))
	if err != nil || json.Unmarshal(b, &inv) != nil || !strings.HasPrefix(inv.Key, "tskey-") {
		return inv, errors.New("Convite incompleto ou corrompido. Peça um novo.")
	}
	if ip, err := netip.ParseAddr(inv.Host); err != nil || !ip.Is4() {
		return inv, errors.New("Convite incompleto ou corrompido. Peça um novo.")
	}
	return inv, nil
}

// tunnelState is shown in Configurações → Conexão pela internet.
type tunnelState struct {
	mu    sync.Mutex
	state string // off | connecting | online | error
	ip    string
	err   string
	ts    *tsnet.Server
	turn  *turn.Server
	lns   []net.Listener
	ready chan struct{} // closed once the guest tunnel is up
}

func (t *tunnelState) set(state, ip, err string) {
	t.mu.Lock()
	t.state, t.ip, t.err = state, ip, err
	t.mu.Unlock()
}

func (t *tunnelState) close() {
	t.mu.Lock()
	ts, tr, lns := t.ts, t.turn, t.lns
	t.ts, t.turn, t.lns = nil, nil, nil
	t.state, t.ip, t.err = "off", "", ""
	t.mu.Unlock()
	for _, ln := range lns {
		ln.Close()
	}
	if tr != nil {
		tr.Close()
	}
	if ts != nil {
		ts.Close()
	}
}

func discardLogs(string, ...any) {}

func newNode(dir, hostname, key string, ephemeral bool) *tsnet.Server {
	return &tsnet.Server{
		Dir: dir, Hostname: hostname, AuthKey: key, Ephemeral: ephemeral,
		Logf: discardLogs, UserLogf: discardLogs,
	}
}

func friendlyTSError(err error) string {
	msg := err.Error()
	switch {
	case strings.Contains(msg, "invalid key"), strings.Contains(msg, "authkey"), strings.Contains(msg, "expired"):
		return "A chave do Tailscale é inválida ou expirou. Gere uma nova."
	case errors.Is(err, context.DeadlineExceeded):
		return "Não foi possível falar com o Tailscale. Confira a internet e tente de novo."
	}
	return "Falha ao conectar: " + msg
}

func shortID() string {
	h, _ := os.Hostname()
	h = strings.ToLower(h)
	var b strings.Builder
	for _, r := range h {
		if (r >= 'a' && r <= 'z') || (r >= '0' && r <= '9') {
			b.WriteRune(r)
		}
	}
	if b.Len() == 0 {
		return "pc"
	}
	return b.String()
}

// ---------------------------------------------------------------- host side

// startInternet joins the host's tailnet and serves the chat server and the
// TURN relay there. Runs in the background; progress shows in InternetStatus.
func (a *App) startInternet() {
	key := a.secret("tailscale-key.bin")
	a.mu.Lock()
	srv := a.host.srv
	a.mu.Unlock()
	t := &a.inet
	if key == "" || srv == nil {
		t.set("error", "", "Ligue o servidor embutido e informe a chave do Tailscale.")
		return
	}
	t.close()
	t.set("connecting", "", "")
	go func() {
		node := newNode(a.path("tailscale-host"), "babutela-"+shortID(), key, false)
		ctx, cancel := context.WithTimeout(context.Background(), 60*time.Second)
		defer cancel()
		if _, err := node.Up(ctx); err != nil {
			node.Close()
			t.set("error", "", friendlyTSError(err))
			return
		}
		web, err := node.Listen("tcp", ":3210")
		if err != nil {
			node.Close()
			t.set("error", "", "Falha ao abrir a porta do servidor no Tailscale.")
			return
		}
		relayLn, err := node.Listen("tcp", ":3478")
		if err != nil {
			web.Close()
			node.Close()
			t.set("error", "", "Falha ao abrir a porta de chamadas no Tailscale.")
			return
		}
		relay, err := startTURN(relayLn, a.turnSecret(), 50100, 50199)
		if err != nil {
			web.Close()
			relayLn.Close()
			node.Close()
			t.set("error", "", "Falha ao iniciar o relé de chamadas.")
			return
		}
		srv.Serve(web)
		ip4, _ := node.TailscaleIPs()
		t.mu.Lock()
		t.ts, t.turn, t.lns = node, relay, []net.Listener{web, relayLn}
		t.state, t.ip, t.err = "online", ip4.String(), ""
		t.mu.Unlock()
	}()
}

func (a *App) stopInternet() { a.inet.close() }

// ---------------------------------------------------------------- friend side

// startGuest joins the host's tailnet with the invite and forwards the local
// ports the page and calls use. The ports open right away; connections wait
// for the tunnel to come up.
func (a *App) startGuest() {
	inv, err := decodeInvite(a.secret("invite.bin"))
	t := &a.guest
	t.close()
	if err != nil {
		t.set("error", "", err.Error())
		return
	}
	ready := make(chan struct{})
	t.mu.Lock()
	t.ready = ready
	t.mu.Unlock()
	t.set("connecting", "", "")
	node := newNode(a.path("tailscale-guest"), "babutela-amigo-"+shortID(), inv.Key, true)

	forwards := []struct{ local, remote string }{
		{guestHTTPAddr, net.JoinHostPort(inv.Host, "3210")},
		{guestTURNAddr, net.JoinHostPort(inv.Host, "3478")},
	}
	var lns []net.Listener
	for _, f := range forwards {
		ln, err := net.Listen("tcp", f.local)
		if err != nil {
			for _, l := range lns {
				l.Close()
			}
			t.set("error", "", fmt.Sprintf("A porta %s já está em uso por outro programa.", f.local))
			return
		}
		lns = append(lns, ln)
		go forward(ln, node, f.remote, ready)
	}
	t.mu.Lock()
	t.ts, t.lns = node, lns
	t.mu.Unlock()

	go func() {
		ctx, cancel := context.WithTimeout(context.Background(), 60*time.Second)
		defer cancel()
		if _, err := node.Up(ctx); err != nil {
			t.set("error", "", friendlyTSError(err))
			return
		}
		ip4, _ := node.TailscaleIPs()
		t.set("online", ip4.String(), "")
		close(ready)
	}()
}

func (a *App) stopGuest() { a.guest.close() }

func forward(ln net.Listener, node *tsnet.Server, remote string, ready <-chan struct{}) {
	for {
		c, err := ln.Accept()
		if err != nil {
			return
		}
		go func() {
			defer c.Close()
			select {
			case <-ready:
			case <-time.After(30 * time.Second):
				return
			}
			ctx, cancel := context.WithTimeout(context.Background(), 15*time.Second)
			r, err := node.Dial(ctx, "tcp", remote)
			cancel()
			if err != nil {
				log.Println("túnel:", err)
				return
			}
			defer r.Close()
			done := make(chan struct{}, 2)
			go func() { io.Copy(r, c); done <- struct{}{} }()
			go func() { io.Copy(c, r); done <- struct{}{} }()
			<-done
		}()
	}
}

// ---------------------------------------------------------------- secrets on disk (DPAPI)

func (a *App) secret(name string) string {
	b, err := os.ReadFile(a.path(name))
	if err != nil {
		return ""
	}
	plain, err := unprotect(b)
	if err != nil {
		return ""
	}
	return string(plain)
}

func (a *App) setSecret(name, value string) error {
	if value == "" {
		os.Remove(a.path(name))
		return nil
	}
	enc, err := protect([]byte(value))
	if err != nil {
		return err
	}
	return os.WriteFile(a.path(name), enc, 0o600)
}
