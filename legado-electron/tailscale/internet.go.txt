//go:build !public

package main

import (
	"errors"
	"strings"
)

// Methods the page calls for Configurações → Conexão pela internet and for
// "Entrar com convite de um amigo" (see frontend/native.js).

func (a *App) isGuest() bool { return a.Settings().Mode == "guest" }

// serverURL is where the page talks to: the friend's server through the
// tunnel in guest mode, the configured address otherwise.
func (a *App) serverURL() string {
	if a.isGuest() {
		return guestServerURL
	}
	return a.Settings().ServerURL
}

// turnURL is the relay the page's calls use, or "" when there is none.
func (a *App) turnURL() string {
	if a.isGuest() {
		return turnURLGuest
	}
	a.mu.Lock()
	defer a.mu.Unlock()
	if a.localTURN != nil {
		return turnURLLocal
	}
	return ""
}

func stateOf(t *tunnelState) (string, string, string) {
	t.mu.Lock()
	defer t.mu.Unlock()
	state := t.state
	if state == "" {
		state = "off"
	}
	return state, t.ip, t.err
}

func (a *App) InternetStatus() map[string]any {
	s := a.Settings()
	t := &a.inet
	if s.Mode == "guest" {
		t = &a.guest
	}
	state, ip, errMsg := stateOf(t)
	return map[string]any{
		"mode": modeOf(s), "internet": s.Internet, "state": state, "ip": ip, "error": errMsg,
		"hasKey": a.secret("tailscale-key.bin") != "", "available": embeddedServer(),
	}
}

func modeOf(s Settings) string {
	if s.Mode == "guest" {
		return "guest"
	}
	return "host"
}

// SetInternet turns internet connections on or off. A non-empty key replaces
// the saved Tailscale auth key.
func (a *App) SetInternet(enabled bool, key string) (map[string]any, error) {
	key = strings.TrimSpace(key)
	if key != "" {
		if !strings.HasPrefix(key, "tskey-") {
			return nil, errors.New("Essa não parece uma chave do Tailscale (ela começa com tskey-).")
		}
		if err := a.setSecret("tailscale-key.bin", key); err != nil {
			return nil, errors.New("Não foi possível guardar a chave.")
		}
	}
	if enabled && a.secret("tailscale-key.bin") == "" {
		return nil, errors.New("Cole a chave do Tailscale para ativar.")
	}
	a.mu.Lock()
	a.settings.Internet = enabled
	a.saveSettings()
	a.mu.Unlock()
	if enabled {
		a.startInternet()
	} else {
		a.stopInternet()
	}
	return a.InternetStatus(), nil
}

// CreateInvite packs the auth key and this computer's tailnet address.
func (a *App) CreateInvite() (string, error) {
	state, ip, _ := stateOf(&a.inet)
	if state != "online" || ip == "" {
		return "", errors.New("Ative a conexão pela internet e espere ela ficar online.")
	}
	key := a.secret("tailscale-key.bin")
	if key == "" {
		return "", errors.New("Falta a chave do Tailscale.")
	}
	return encodeInvite(invite{Key: key, Host: ip}), nil
}

// JoinInvite switches this app to the friend's server. The page reloads
// afterwards to pick up the new address.
func (a *App) JoinInvite(code string) error {
	if _, err := decodeInvite(code); err != nil {
		return err
	}
	if err := a.setSecret("invite.bin", strings.Join(strings.Fields(code), "")); err != nil {
		return errors.New("Não foi possível guardar o convite.")
	}
	a.stopInternet()
	a.stopHost()
	a.mu.Lock()
	a.settings.Mode = "guest"
	a.saveSettings()
	a.mu.Unlock()
	a.startGuest()
	return nil
}

// LeaveGuest goes back to hosting (or the configured address).
func (a *App) LeaveGuest() {
	a.stopGuest()
	a.setSecret("invite.bin", "")
	a.mu.Lock()
	a.settings.Mode = ""
	a.saveSettings()
	s := a.settings
	a.mu.Unlock()
	if s.HostServer {
		a.startHost()
		if s.Internet {
			a.startInternet()
		}
	}
}
